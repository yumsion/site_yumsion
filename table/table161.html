sh ping_or_serivce_check.sh #!/bin/bash

trap 'exit 130' INT
# TCP端口检测脚本
# 功能：监控指定IP的TCP端口状态，检测到持续10分钟不可用时发送通知

# 配置文件（可选）
LOG_FILE="/user01/log/tcp_port_monitor.log"
MSG_LOG_FILE="/user01/log/port_notification.log"
#离线触发告警的时间,当设备离线超过这个时候才告警,避免重启告警
ALERT_TRIGGER_SECONDS=120
#多长时间检测一次
CHECK_INTERVAL_SECONDS=60
#
WEBHOOK_MSG=""

# 初始化监控数据
declare -A DEVICE_CURRENT_STATUS # 当前状态（1-可用，0-不可用）
declare -A DEVICE_FAIL_TIME # 开始不可用时间
declare -A DEVICE_FAIL_CONTER # 错误次数计数器,用于检测频繁掉线的问题
declare -A DEVICE_NOTIFIED # 是否已发送通知（1-已发送，0-未发送）

# 要检测的设备列表（格式：IP:端口/NULL:CMD函数名称：分类描述）
DEVICES=(
"192.168.187.114:NULL:PING_CHECK:H3C_AP"
"172.18.2.5:NULL:PING_CHECK:UNKNOWN"
"172.18.12.4:80:PORT_CHECK:CAMERA"
"172.30.240.21:80:PORT_CHECK:CAMERA"
)

# 定义MSG_SEND函数
MSG_SEND() {
local ip=$1
local port=$2
local status=$3
local timestamp=$(date '+%Y-%m-%d %H:%M:%S')

# 构造通知消息
local message="[$timestamp] 告警：设备 $ip 的端口 $port 状态变为 $status"

# 模拟发送通知（这里可以替换为实际的通知方式）
# 例如：发送邮件、调用API、写入日志等

# 示例1：写入通知日志文件
echo "$message" >> "$MSG_LOG_FILE"

# 示例2：发送到syslog
# logger -t "port_monitor" "$message"

# 示例3：发送HTTP请求到webhook
# curl -X POST -H "Content-Type: application/json" -d "{\"text\":\"$message\"}" https://your-webhook-url

WEBHOOK_MSG="$WEBHOOK_MSG
$timestamp CAMERA $ip STATUS CHANGE TO $status (检测摄像头总量:${#DEVICES[@]}）"
# 示例4：发送邮件
# echo "$message" | mail -s "端口监控告警" admin@example.com

# 模拟发送成功率（实际使用时请删除此逻辑）
# 这里简单模拟：90%成功率用于测试，实际使用时直接返回0
# 生产环境请删除以下模拟代码，直接返回0表示成功
zabbix_sender -z 172.16.6.1 -i - <<_EOF
172.16.1.9-AMG9 amg9_fallback $timestamp CAMERA $ip STATUS CHANGE TO $status (检测摄像头总量:${#DEVICES[@]})
_EOF
if [ $? -ne 0 ]; then
echo "[$timestamp] zabbix通知发送失败: $message" >> "$LOG_FILE"
return 1 # 模拟发送失败
fi

echo "[$timestamp] 通知发送成功: $message" >> "$LOG_FILE"
return 0 # 模拟发送成功
}

# 定义检测函数
function PORT_CHECK() {
local ip=$1
local port=$2
nc -v -z -n -w 2 $ip $port 1>/dev/null 2>&1
return $?
}
function PING_CHECK() {
local ip=$1
local port=$2
ping -w 2 -c 1 $ip 1>/dev/null 2>&1
return $?
}

function DNS_CHECK() {
local ip=$1
local port=$2
timeout -s SIGINT 1 dig www.feishu.cn @$ip | grep --fixed-strings $'IN\tA\t' 1>/dev/null 2>&1
#timeout -s SIGINT 1 dig www.feishu.cn @$ip | grep --fixed-strings $'IN\tA\t' 1>/dev/null 2>&1
return $?
}

# 初始化所有设备状态为可用
for device in "${DEVICES[@]}"; do
DEVICE_CURRENT_STATUS["$device"]=1 # 1=可用，0=不可用
DEVICE_FAIL_TIME["$device"]=""
DEVICE_FAIL_CONTER["$device"]=0
DEVICE_NOTIFIED["$device"]=0
done

echo "TCP端口监控脚本启动..."
echo "监控设备列表："
for device in "${DEVICES[@]}"; do
echo " - $device"
done
echo "日志文件: $LOG_FILE"
echo "通知日志: $MSG_LOG_FILE"
echo "检查间隔: $CHECK_INTERVAL_SECONDS + ，持续$ALERT_TRIGGER_SECONDS秒不可用才发送通知"
echo "----------------------------------------"

#MAIN
# 主循环
bash ./feishu_msg_send.sh "设备离线检测脚本启动, (检测设备总量:${#DEVICES[@]}）" AMG9_YUM_SW_OFFLINE_CHECK
while true; do
current_time=$(date '+%Y-%m-%d %H:%M:%S')

for device in "${DEVICES[@]}"; do
#ip=$(echo $device | cut -d':' -f1)
#port=$(echo $device | cut -d':' -f2)
printf "\r%-${COLUMNS}s" "$device"
IFS=: read ip port CHECK_CMD DEVICE_TYPE _<<< "$device"

# 检测端口状态

$CHECK_CMD "$ip" "$port"
if [[ $? -eq 0 ]] ; then
# 端口可用
if [ ${DEVICE_CURRENT_STATUS["$device"]} -eq 0 ]; then
# 状态从不可用变为可用
echo "[$current_time] $DEVICE_TYPE $device - 状态变化: 不可用 → 可用" >> $LOG_FILE
DEVICE_CURRENT_STATUS["$device"]=1
DEVICE_FAIL_TIME["$device"]=""
DEVICE_NOTIFIED["$device"]=0
fi
else
# 端口不可用
if [ ${DEVICE_CURRENT_STATUS["$device"]} -eq 1 ]; then
# 状态从可用变为不可用，记录开始时间
echo "[$current_time] $DEVICE_TYPE $device - 状态变化: 可用 → 不可用" >> $LOG_FILE
DEVICE_CURRENT_STATUS["$device"]=0
DEVICE_FAIL_TIME["$device"]="$current_time"
((DEVICE_FAIL_CONTER["$device"]++))
DEVICE_NOTIFIED["$device"]=0
elif [ ${DEVICE_CURRENT_STATUS["$device"]} -eq 0 ]; then
# 已经是不可用状态，检查持续时间
if [ -n "${DEVICE_FAIL_TIME["$device"]}" ] && [ ${DEVICE_NOTIFIED["$device"]} -eq 0 ]; then
# 计算不可用持续时间（分钟）
fail_epoch=$(date -d "${DEVICE_FAIL_TIME["$device"]}" +%s)
current_epoch=$(date +%s)
duration_seconds=$(( current_epoch - fail_epoch ))

if [ $duration_seconds -ge $ALERT_TRIGGER_SECONDS ]; then
echo "[$current_time] $DEVICE_TYPE $device - 已持续不可用 ${duration_seconds} 秒，准备发送通知"
# 尝试发送通知
if MSG_SEND "$ip" "$port" "$DEVICE_TYPE $CHECK_CMD 不可用(持续${duration_seconds}秒,总错误次数:${DEVICE_FAIL_CONTER[$device]})"; then
# 发送成功，标记已发送通知
DEVICE_NOTIFIED["$device"]=1
echo "[$current_time] $DEVICE_TYPE $device - 通知发送成功，已标记"
else
echo "[$current_time] $DEVICE_TYPE $device - 通知发送失败，等待下次重试"
# 不改变NOTIFIED状态，下次继续尝试发送
fi
else
# 未达到10分钟，记录当前持续时间
echo "[$current_time] $DEVICE_TYPE $device - 已不可用 ${duration_seconds} 秒，未达到通知阈值($ALERT_TRIGGER_SECONDS 秒)"
fi
fi
fi
fi

# 记录当前状态到日志（可选）
if [ ${DEVICE_CURRENT_STATUS["$device"]} -eq 1 ]; then
status_text="可用"
else
status_text="不可用"

REPORT="${REPORT}
[$current_time] $device - 当前状态: $status_text"
fi
done

echo "$REPORT"

if [[ -n "${WEBHOOK_MSG}" ]]
then
bash ./feishu_msg_send.sh "$WEBHOOK_MSG
$REPORT" "AMG9_YUM_SW_OFFLINE_CHECK"
if [ $? -ne 0 ]; then
echo "[$timestamp] feishu通知发送失败: $message" >> "$LOG_FILE"
return 1 # 模拟发送失败
fi

fi


REPORT=""
WEBHOOK_MSG=""


# 等待60秒后再次检测
echo "sleep $CHECK_INTERVAL_SECONDS"
sleep $CHECK_INTERVAL_SECONDS
done


exit
