return 403;
}

proxy_set_header Host $host;
proxy_set_header X-Real-IP $remote_addr;
proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
proxy_set_header X-Forwarded-Proto $scheme;
}
}
}

map 用正则 ~^/agc/ 匹配 URI 前缀。map 是「按需触发」的，只有当你真正引用了 $is_allowed 这个变量时，Nginx 才会去计算它。
命中白名单 $is_allowed=1，否则为 0，返回 403。
新增白名单只需在 map 里加一行，不用改 location。
注意：if 在 location 中一般被认为是「邪恶的」，但这里只做 return，是官方推荐的少数安全用法之一。

#源地址中使用CIDR
在 Nginx 的 map 指令中，默认情况下是不支持 CIDR（如 /24）格式的。
map $remote_addr $upstream05 {
192.168.140.0/24 $upstream04;
172.16.6.9 $upstream04;
default $upstream01;
}

Nginx 的 map 指令在处理第一个参数（匹配源）时，默认将其视为字符串或正则表达式。
当你写 192.168.140.0/24 时，Nginx 会把它当成一个普通的字符串去匹配。
由于 $remote_addr 的值是具体的 IP（如 192.168.140.14），它和字符串 "192.168.140.0/24" 显然不相等。
因此，匹配失败，逻辑直接滑向了 default。

如果你想在 map 中使用 CIDR 格式，你必须显式地使用 geo 模块，或者在 map 中使用正则表达式。
方案 A：使用 geo 指令（最推荐，性能最好）
geo 专门用于将 IP 地址映射为变量，且天然支持 CIDR 格式。

stream {
# 1. 先用 geo 定义 IP 归属组
geo $client_group {
192.168.140.0/24 internal_network;
172.16.6.9/32 internal_network;
default external_network;
}

# 2. 根据分组来选择对应的 upstream
map $client_group $upstream05 {
internal_network $upstream04;
default $upstream01;
}

# ... 其余配置
}
