获取指定IP通讯的进程 查询进程脚本
# 精简监控版本（修正版）
# 文件名: MonitorSimple.ps1

$targetIP = "4.3.2.1"
$outputFile = "D:\1.txt"

Write-Host "监控连接到 $targetIP 的连接，每10秒检查一次..." -ForegroundColor Cyan
Write-Host "按 Ctrl+C 停止`n"

while ($true) {
$connections = netstat -ano | Select-String $targetIP

if ($connections) {
$timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
$processIds = $connections | ForEach-Object {
if ($_ -match '\s+(\d+)\s*$') { $matches[1] }
} | Select-Object -Unique

# 构建输出
"`n========== $timestamp ==========" | Out-File $outputFile -Append -Encoding UTF8
$connections | Out-File $outputFile -Append -Encoding UTF8

foreach ($procId in $processIds) {
try {
$proc = Get-Process -Id $procId -ErrorAction Stop
"PID: $procId -> $($proc.ProcessName) ($($proc.Path))" | Out-File $outputFile -Append -Encoding UTF8
}
catch {
"PID: $procId -> 无法获取进程信息" | Out-File $outputFile -Append -Encoding UTF8
}
}

Write-Host "[$timestamp] 发现连接，已写入 $outputFile" -ForegroundColor Green
}

Start-Sleep -Seconds 10
}
