#!/bin/bash
# ==============================================================================
# 脚本名称: check_conntrack_tcp_dual.sh
# 说明: 全量双向对比 TCP 协议栈与 Conntrack 参数 (兼顾“连接连通性”与“抗爆表能力”)
# ==============================================================================

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[0;33m'
BLUE='\033[0;34m'
NC='\033[0m'

echo -e "${BLUE}======================================================================${NC}"
echo -e "${BLUE} Linux 全量 TCP vs Conntrack 双向协同与资源安全性诊断工具 ${NC}"
echo -e "${BLUE}======================================================================${NC}"
echo

if ! lsmod | grep -q nf_conntrack; then
echo -e "${RED}[ERROR] nf_conntrack 模块未加载！请先执行 modprobe nf_conntrack${NC}"
exit 1
fi

get_sysctl() {
local val
val=$(sysctl -n "$1" 2>/dev/null)
echo "${val:-0}"
}

# 1. 提取 Conntrack 参数
ct_est=$(get_sysctl net.netfilter.nf_conntrack_tcp_timeout_established)
ct_close_wait=$(get_sysctl net.netfilter.nf_conntrack_tcp_timeout_close_wait)
ct_fin_wait=$(get_sysctl net.netfilter.nf_conntrack_tcp_timeout_fin_wait)
ct_time_wait=$(get_sysctl net.netfilter.nf_conntrack_tcp_timeout_time_wait)
ct_syn_sent=$(get_sysctl net.netfilter.nf_conntrack_tcp_timeout_syn_sent)
ct_syn_recv=$(get_sysctl net.netfilter.nf_conntrack_tcp_timeout_syn_recv)
ct_max=$(get_sysctl net.netfilter.nf_conntrack_max)
ct_count=$(get_sysctl net.netfilter.nf_conntrack_count)

# 精确获取 hashsize
hashsize=0
if [ -f /sys/module/nf_conntrack/parameters/hashsize ]; then
hashsize=$(cat /sys/module/nf_conntrack/parameters/hashsize)
elif [ -f /proc/sys/net/netfilter/nf_conntrack_buckets ]; then
hashsize=$(cat /proc/sys/net/netfilter/nf_conntrack_buckets)
fi

# 2. 提取 TCP 关键参数
tcp_fin_timeout=$(get_sysctl net.ipv4.tcp_fin_timeout)
tcp_tw_reuse=$(get_sysctl net.ipv4.tcp_tw_reuse)
tcp_keepalive_time=$(get_sysctl net.ipv4.tcp_keepalive_time)
tcp_keepalive_intvl=$(get_sysctl net.ipv4.tcp_keepalive_intvl)
tcp_keepalive_probes=$(get_sysctl net.ipv4.tcp_keepalive_probes)
tcp_syn_retries=$(get_sysctl net.ipv4.tcp_syn_retries)
tcp_synack_retries=$(get_sysctl net.ipv4.tcp_synack_retries)

# 计算 Keepalive 总探测尝试周期
keepalive_total_fail_time=$((tcp_keepalive_intvl * tcp_keepalive_probes))

# 打印提取数据
echo -e "${GREEN}[1] 当前网络参数汇总提取表:${NC}"
echo "----------------------------------------------------------------------"
printf "%-42s : %-10s | %-32s : %s\n" "net.ipv4.tcp_fin_timeout" "${tcp_fin_timeout}s" "nf_conntrack_tcp_timeout_fin_wait" "${ct_fin_wait}s"
printf "%-42s : %-10s | %-32s : %s\n" "net.ipv4.tcp_tw_reuse" "${tcp_tw_reuse}" "nf_conntrack_tcp_timeout_time_wait" "${ct_time_wait}s"
printf "%-42s : %-10s | %-32s : %s\n" "net.ipv4.tcp_keepalive_time" "${tcp_keepalive_time}s" "nf_conntrack_tcp_timeout_established" "${ct_est}s"
printf "%-42s : %-10s | %-32s : %s\n" "keepalive intvl*probes (探测失败时长)" "${keepalive_total_fail_time}s" "nf_conntrack_tcp_timeout_close_wait" "${ct_close_wait}s"
printf "%-42s : %-10s | %-32s : %s\n" "net.ipv4.tcp_syn_retries" "${tcp_syn_retries}次" "nf_conntrack_tcp_timeout_syn_sent" "${ct_syn_sent}s"
printf "%-42s : %-10s | %-32s : %s\n" "nf_conntrack_hashsize" "${hashsize}" "nf_conntrack_count / max" "${ct_count} / ${ct_max}"
echo "----------------------------------------------------------------------"
echo

# 3. 诊断评估
echo -e "${GREEN}[2] 双向逻辑匹配与系统风险评估:${NC}"
ERR_COUNT=0
WARN_COUNT=0

# ------------------------------------------------------------------------------
# 规则 1: FIN_WAIT 状态匹配 (下限: 防误杀; 上限: 防白白占用表项)
# ------------------------------------------------------------------------------
if [ "$ct_fin_wait" -lt "$tcp_fin_timeout" ]; then
echo -e "${RED}[HIGH RISK] 规则 1 校验失败 (FIN_WAIT 阶段 Conntrack 过早回收):${NC}"
echo -e " └─ 数值: ct_fin_wait (${ct_fin_wait}s) < tcp_fin_timeout (${tcp_fin_timeout}s)"
echo -e " └─ 隐患: Conntrack 表项先被删除，后续 FIN 包或 ACK 包会被标记为 INVALID 并丢弃。"
echo -e " └─ 建议: 将 ct_fin_wait 调整为 ≥ tcp_fin_timeout (推荐值均设为 30s)。"
((ERR_COUNT++))
elif [ "$((ct_fin_wait - tcp_fin_timeout))" -gt 30 ]; then
echo -e "${YELLOW}[WARNING] 规则 1 触发告警 (FIN_WAIT 阶段 Conntrack 表项挂死时间过长):${NC}"
echo -e " └─ 数值: ct_fin_wait (${ct_fin_wait}s) 远大于 tcp_fin_timeout (${tcp_fin_timeout}s)"
echo -e " └─ 隐患: Socket 已经彻底销毁，但 Conntrack 表项还要挂手长时间，白白浪费内存容量。"
echo -e " └─ 建议: 将 ct_fin_wait 调低至与 tcp_fin_timeout 一致 (推荐 30s)。"
((WARN_COUNT++))
else
echo -e "[PASS] 规则 1 检查通过: FIN_WAIT 状态回收时间精准协同 (${ct_fin_wait}s vs ${tcp_fin_timeout}s)。"
fi

# ------------------------------------------------------------------------------
# 规则 2: TIME_WAIT 状态匹配 (下限: 防端口重用时 SYN INVALID 丢包; 上限: 防表项积累)
# ------------------------------------------------------------------------------
if [ "$tcp_tw_reuse" -ge 1 ] && [ "$ct_time_wait" -gt 60 ]; then
echo -e "${YELLOW}[WARNING] 规则 2 触发告警 (TIME_WAIT 快速重用与 Conntrack 超时冲突):${NC}"
echo -e " └─ 数值: tcp_tw_reuse = ${tcp_tw_reuse} (已开启端口复用)，ct_time_wait = ${ct_time_wait}s (>60s)"
echo -e " └─ 隐患: TCP 栈已允许复用端口建连，但 Conntrack 依旧保留旧条目，新 SYN 包可能被认作 INVALID 丢弃！"
echo -e " └─ 建议: 开启 tw_reuse 时，务必将 ct_time_wait 降至 30s。"
((WARN_COUNT++))
elif [ "$ct_time_wait" -gt 60 ]; then
echo -e "${YELLOW}[WARNING] 规则 2 触发告警 (TIME_WAIT 表项残留时间过长):${NC}"
echo -e " └─ 数值: ct_time_wait = ${ct_time_wait}s"
echo -e " └─ 隐患: 高并发短连接场景下，表项回收太慢，极易导致 Conntrack 爆表。"
echo -e " └─ 建议: 将 ct_time_wait 降至 30s。"
((WARN_COUNT++))
else
echo -e "[PASS] 规则 2 检查通过: TIME_WAIT 状态的 Conntrack 回收配置合理 (${ct_time_wait}s)。"
fi

# ------------------------------------------------------------------------------
# 规则 3: ESTABLISHED 状态双向校验 (下限: 防心跳前切断; 上限: 防僵尸连接霸满表项)
# ------------------------------------------------------------------------------
if [ "$tcp_keepalive_time" -ge "$ct_est" ]; then
echo -e "${RED}[HIGH RISK] 规则 3 校验失败 (KeepAlive 心跳周期的首次探测慢于 Conntrack 超时):${NC}"
echo -e " └─ 数值: tcp_keepalive_time (${tcp_keepalive_time}s) ≥ ct_est (${ct_est}s)"
echo -e " └─ 隐患: 在 TCP 触发第一次保活心跳之前，Conntrack 表项就已经失效，导致连接被防火墙静默断掉。"
echo -e " └─ 建议: 确保 tcp_keepalive_time < ct_est。"
((ERR_COUNT++))
elif [ "$ct_est" -gt 14400 ]; then
echo -e "${YELLOW}[WARNING] 规则 3 触发告警 (ESTABLISHED 超时过长，有严重爆表隐患):${NC}"
echo -e " └─ 数值: ct_est = ${ct_est}s (约 $((ct_est/3600)) 小时)，tcp_keepalive_time = ${tcp_keepalive_time}s"
echo -e " └─ 隐患: 虽然能够成功保活，但超长的超时会导致大量非正常断网的“僵尸连接”死扣内存空间不释放！"
echo -e " └─ 建议: 将 ct_est 降低至 7200s (2小时) 或更短 (如 1200s~3600s)。"
((WARN_COUNT++))
else
echo -e "[PASS] 规则 3 检查通过: ESTABLISHED 保持周期合理 (${ct_est}s)，且大于 KeepAlive 首次探测时间 (${tcp_keepalive_time}s)。"
fi

# ------------------------------------------------------------------------------
# 规则 4: CLOSE_WAIT 状态校验 (防应用层 Socket 未释放/连接池泄露)
# ------------------------------------------------------------------------------
if [ "$ct_close_wait" -gt 30 ]; then
echo -e "${YELLOW}[WARNING] 规则 4 触发告警 (CLOSE_WAIT 处于高风险耗尽区):${NC}"
echo -e " └─ 数值: ct_close_wait = ${ct_close_wait}s (>30s)"
echo -e " └─ 隐患: 若应用层遇到异常未及时 close() 释放 Socket，Conntrack 表将长时间充斥半关闭连接直至爆表。"
echo -e " └─ 建议: 将 ct_close_wait 降低至 30s 以下。"
((WARN_COUNT++))
else
echo -e "[PASS] 规则 4 检查通过: ct_close_wait (${ct_close_wait}s) 控制在安全可控范围。"
fi

# ------------------------------------------------------------------------------
# 规则 5: SYN_SENT 状态校验 (下限: 防重试被丢弃; 上限: 防 SYN 攻击/无法到达的 IP 挂死)
# ------------------------------------------------------------------------------
# 计算 SYN 重试实际退避总耗时 (1s + 2s + 4s + ...)
est_syn_time=0
for ((i=0; i<tcp_syn_retries; i++)); do
est_syn_time=$((est_syn_time + (1 << i)))
done

if [ "$ct_syn_sent" -lt "$est_syn_time" ]; then
echo -e "${RED}[HIGH RISK] 规则 5 校验失败 (SYN_SENT 超时短于 TCP 重试周期):${NC}"
echo -e " └─ 数值: ct_syn_sent (${ct_syn_sent}s) < 理论 SYN 重试时间 (${est_syn_time}s)"
echo -e " └─ 隐患: 防火墙提前清掉了握手表项，导致后续发出的 SYN 重发包被当成 INVALID 丢弃。"
echo -e " └─ 建议: 将 tcp_syn_retries 降至 3~4 次，或调大 ct_syn_sent。"
((ERR_COUNT++))
elif [ "$ct_syn_sent" -gt 60 ]; then
echo -e "${YELLOW}[WARNING] 规则 5 触发告警 (SYN_SENT 保持时间过长):${NC}"
echo -e " └─ 数值: ct_syn_sent = ${ct_syn_sent}s (>60s)"
echo -e " └─ 隐患: 遇到网络不可达或 SYN Flood 攻击时，半开连接条目滞留过久，容易挤爆 Conntrack 表。"
echo -e " └─ 建议: 将 ct_syn_sent 降至 30s，同时控制 tcp_syn_retries 次数为 3~4 次。"
((WARN_COUNT++))
else
echo -e "[PASS] 规则 5 检查通过: SYN_SENT 超时设定合理 (${ct_syn_sent}s)，与 SYN 重试机制完好适配。"
fi

# ------------------------------------------------------------------------------
# 规则 6: Hashsize 与 Max 配比区间校验 (防 CPU 软中断飙高或内存浪费)
# ------------------------------------------------------------------------------
if [ "$hashsize" -gt 0 ]; then
ratio=$((ct_max / hashsize))
if [ "$ratio" -gt 8 ]; then
echo -e "${RED}[HIGH RISK] 规则 6 校验失败 (Conntrack 桶深度配比严重失衡):${NC}"
echo -e " └─ 数值: ct_max = ${ct_max}, hashsize = ${hashsize} (当前配比 ${ratio}:1 > 8:1)"
echo -e " └─ 隐患: Hash 桶过浅导致冲突链表太长！并发升高时 CPU 软中断 (softirq) 会直接飙满卡死。"
echo -e " └─ 建议: 调整 hashsize，使配比保持在 4:1 (即 hashsize = ct_max / 4)。"
((ERR_COUNT++))
elif [ "$ratio" -lt 2 ]; then
echo -e "${YELLOW}[WARNING] 规则 6 触发告警 (Hash 桶分配过多，占用额外连续物理内存):${NC}"
echo -e " └─ 数值: ct_max = ${ct_max}, hashsize = ${hashsize} (当前配比 ${ratio}:1)"
echo -e " └─ 隐患: 配比过小虽然查找快，但会白白消耗 Zone Normal 的连续物理内存。"
echo -e " └─ 建议: 推荐最佳黄金配比为 4:1 (例如 max=262144 时，hashsize 设为 65536)。"
((WARN_COUNT++))
else
echo -e "[PASS] 规则 6 检查通过: nf_conntrack_max 与 hashsize 配比为 ${ratio}:1 (处在 2:1 ~ 8:1 黄金区间)。"
fi
else
echo -e "${YELLOW}[WARNING] 规则 6 检查跳过: 未能在系统中获取到有效的 hashsize 值。${NC}"
fi

# 4. 总结与建议
echo
echo -e "${BLUE}======================================================================${NC}"
if [ "$ERR_COUNT" -eq 0 ] && [ "$WARN_COUNT" -eq 0 ]; then
echo -e "${GREEN}总结评估: 完美！你的网络栈参数与 Conntrack 配置在“连通性”与“资源回收”上达到了极佳的平衡。${NC}"
elif [ "$ERR_COUNT" -eq 0 ]; then
echo -e "${YELLOW}总结评估: 发现 $WARN_COUNT 项优化告警（主要集中在资源回收效率上）。无阻断性高危风险，建议根据提示进行调优。${NC}"
else
echo -e "${RED}总结评估: 检测到 $ERR_COUNT 项高危参数冲突及 $WARN_COUNT 项告警！请优先修复高危项，防止出现连接误杀或服务挂死！${NC}"
fi
echo -e "${BLUE}======================================================================${NC}"

cat /proc/net/stat/nf_conntrack | awk '
NR>1 {
drop += hex2dec($2);
early_drop += hex2dec($3);
insert_failed += hex2dec($5);
}
function hex2dec(h) { return sprintf("%d", "0x" h) }
END {
print "Conntrack 丢包总数 (drop) : " drop;
print "表满早丢包数 (early_drop) : " early_drop;
print "插入哈希表失败数 (insert_failed) : " insert_failed;
}'

nstat / /proc/net/netstat
当 Conntrack 和 TCP 协同出问题时（如 Conntrack 先超时回收了表项，后续的 ACK/FIN 包被当成无效包），TCP 层会记录相应的丢包与乱序统计。
netstat -s | grep -i -E "invalid|dropped|reset|pruned|overflow"

localedef的作用：
localedef 是 Linux 系统中用于编译和生成 locale 数据的命令，定义编译器。

-f --charmap，指定字符映射文件（字符编码）
最后一个参数 要生成的 locale 名称
从 /usr/share/i18n/locales/en_US 读取 locale 定义
从 /usr/share/i18n/charmaps/UTF-8.gz 读取 UTF-8 字符编码映射。
将上述信息编译成二进制格式，生成locale文件并注册到系统
