mkfs.btrfs -m raid1 -d raid1 /dev/sdb /dev/sdc /dev/sdd -L mydata
或
mkfs.btrfs -m raid1 -d raid1 \
/dev/disk/by-id/ata-你的8T硬盘ID \
/dev/disk/by-id/ata-你的第一块4T硬盘ID \
/dev/disk/by-id/ata-你的第二块4T硬盘ID \
-L mydata

-m raid1：将元数据（Metadata）设置为 RAID 1（双副本存放，极为重要，防止元数据损坏导致文件系统崩溃）。
-d raid1：将数据（Data）设置为 RAID 1。
-L mydata：给文件系统设置一个标签（Label）。

# 创建挂载点
mkdir -p /mnt/btrfs_pool

# 临时挂载
mount /dev/sdb /mnt/btrfs_pool
mount UUID=你的文件系统UUID /mnt/btrfs_pool
#挂载 Btrfs 多盘文件系统时，你只需要写任意一块属于这个 Btrfs 卷的硬盘即可。
#fstab挂载

vim /etc/fstab
UUID=你的UUID /mnt/btrfs_pool btrfs defaults,compress=zstd 0 0

# 查看空间使用情况
btrfs filesystem df /mnt/btrfs_pool
btrfs filesystem show /mnt/btrfs_pool

#磁盘空间不均时的 Balance（平衡）操作
btrfs balance start /mnt/btrfs_pool

#定期进行文件系统 scrub（数据校验）
RAID 1 虽然有双副本，但如果发生静默数据损坏（Bit Rot），系统需要校验才能知道哪个副本是正确的。定期运行校验：
# 后台启动校验
btrfs scrub start /mnt/btrfs_pool
# 查看校验进度和状态
btrfs scrub status /mnt/btrfs_pool
