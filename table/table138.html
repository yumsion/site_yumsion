git reset #堆栈式删除提交
  1. git reset --mixed HEAD~1 #默认选项，删除最后一次提交，不同步工作区（保留工作区的内容）
  2. git reset --soft HEAD~1 #删除最后一次提交，不同步暂存区（保留暂存区的内容）
  3. git reset --hard HEAD~1 #删除最后一次提交，同步工作区和暂存区（删除工作区和暂存区的内容）
