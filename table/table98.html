function! GetVimsessionFilename()
if expand('%:p') != ''
" 使用当前文件的相同目录，添加 .viminfo 后缀
return expand('%:p') . '.session'
else
" 如果没有文件名，使用默认位置
return
endif
endfunction

function! AutoLoadVimsession()
let vimsession_file = GetVimsessionFilename()
if !empty(vimsession_file)
if filereadable(vimsession_file)
execute 'source ' . fnameescape(vimsession_file)
endif
endif
endfunction

" 自动保存和加载 viminfo 配置（带存在性检查）
function! GetViminfoFilename()
if expand('%:p') != ''
" 使用当前文件的相同目录，添加 .viminfo 后缀
return expand('%:p') . '.viminfo'
else
" 如果没有文件名，使用默认位置
return expand('~/.vim/default.viminfo')
endif
endfunction

function! AutoSaveViminfo()
let viminfo_file = GetViminfoFilename()

" 检查 viminfo 文件是否存在
if filereadable(viminfo_file)
" 使用 ! 强制覆盖已存在的文件
execute 'wviminfo! ' . fnameescape(viminfo_file)
" echo 'Viminfo auto-saved: ' . viminfo_file
else
" 可选：静默不保存，或者显示提示
" echo 'Viminfo file not found, skip saving: ' . viminfo_file
endif
endfunction

function! AutoLoadViminfo()
let viminfo_file = GetViminfoFilename()
if filereadable(viminfo_file)
" 使用 ! 强制覆盖当前寄存器
execute 'rviminfo! ' . fnameescape(viminfo_file)
" echo 'Viminfo auto-loaded: ' . viminfo_file
endif
endfunction

" 自动保存和加载的触发器
autocmd BufReadPost * call AutoLoadViminfo()
autocmd BufWritePost,FocusLost * call AutoSaveViminfo()
autocmd VimEnter * nested call AutoLoadVimsession()
autocmd VimLeave * call AutoSaveViminfo()
