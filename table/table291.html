mysql_import.sh #大文件sql数据导入，逐步导入，避免过多数据导入失败。需要SESSION ADMIN权限来关闭bin log，以加快导入速度。 #!/bin/bash

# 参数配置
MYSQL_BIN='/usr/bin/mysql'
MYSQL_BIN='/usr/bin/mariadb'
MYSQL_CNF="/etc/my.cnf"
MYSQL_HOST="1.1.1.2"
MYSQL_PORT="3306"
MYSQL_USER="user01"
MYSQL_PASSWORD="pass01"
TARGET_DB="db01"
SQL_FILE="sql01.sql"
SQL_LINE_VARIABLES_FILE="/dev/shm/SQL_LINE_VARIABLES_FILE.txt"
BATCH_SIZE=27
BATCH_SIZE_FILE='/dev/shm/count.txt'

# 检查 SQL 文件是否存在
if [ ! -f "$SQL_FILE" ]; then
echo "[ERROR] 错误: 文件 $SQL_FILE 不存在！"
exit 1
fi
# 检查 SQL_LINE_VARIABLES_FILE 文件是否存在
if [[ -f "$SQL_LINE_VARIABLES_FILE" && ${#SQL_LINE_VARIABLES_FILE} -gt 5 ]]; then
rm -rf "$SQL_LINE_VARIABLES_FILE"
fi
touch "$SQL_LINE_VARIABLES_FILE"

echo "[INFO] SQL_FILE $SQL_FILE 存在。"


exec {FD_SQL}<"$SQL_FILE"
[[ $? -ne 0 ]] && echo "[ERROR] $SQL_FILE FAILED, exit..." && exit 1

echo "[INFO] SQL_FILE 文件已打开，准备开始读取。"
TOTAL_COUNTER=0

while true
do
echo "[INFO] $(date) TOTAL_COUNTER : $TOTAL_COUNTER" >&2

#以下内容会输送到MYSQL中，不要再STDOUT中输出非必要内容
{
#存储的变量的初始化
while read -r line || [ -n "$line" ]
do
echo "$line"
done <"$SQL_LINE_VARIABLES_FILE"

# 1. 连接初期设置（关闭约束与日志以提高写入性能）
echo "SET FOREIGN_KEY_CHECKS=0;"
echo "SET UNIQUE_CHECKS=0;"
echo "SET AUTOCOMMIT=0;"
echo "SET sql_log_bin=0;"
echo "BEGIN;"

# 2. 使用 read 逐行读取 SQL 并按批次 COMMIT/BEGIN
count=0
while IFS= read -u "$FD_SQL" -r line || [ -n "$line" ]; do
# echo "$line" >&2
#命令指定行
if [[ "$line" =~ ^\/\*.*\ SET\ .*=.*\*\/ ]] ; then
#echo '[DEBUG] SET LINE MATCH' >&2
line=${line#* SET}
line=${line% \**}
line="SET $line;"
grep -q --fixed-strings "$line" "$SQL_LINE_VARIABLES_FILE"
#注意，此处有瑕疵，没有次序对比，可能有变动的变量导致问题
if [[ $? -ne 0 ]]
then
echo "$line" >>"$SQL_LINE_VARIABLES_FILE"
#echo "[DEBUG] SQL_LINE_VARIABLES+= $line" >&2
fi
echo "$line"
echo "[INFO] SET VARIABLES : $line" >&2
((count++))
continue
fi
if [ $count -eq 0 ]; then
echo "BEGIN;"
echo "$line"
echo "[INFO] 第一行$line" >&2
#避免create语句多行被分割，使用分号确认结尾
elif [[ $count -ge $BATCH_SIZE && "$line" =~ \;[[:space:]]*$ ]]; then
break;
else
echo "$line"
fi
((count++))
#echo "[DEBUG] count $count" >&2
done
echo "$count" >"$BATCH_SIZE_FILE"

# 3. 循环结束后提交剩余未提交的数据并恢复设置
echo "$line"
echo "[INFO] 最后一行$line" >&2
echo "COMMIT;"
echo "SET FOREIGN_KEY_CHECKS=1;"
echo "SET UNIQUE_CHECKS=1;"

} | "$MYSQL_BIN" --defaults-file="$MYSQL_CNF" \
--host="$MYSQL_HOST" \
--port="$MYSQL_PORT" \
--user="$MYSQL_USER" \
--password="password01" \
--max-allowed-packet=1G \
--line-numbers \
--skip-ssl \
--unbuffered \
"$TARGET_DB"

# 获取管道中 mysql 命令的返回状态码
if [[ ${PIPESTATUS[1]} -eq 0 ]]; then
echo "$(cat "$BATCH_SIZE_FILE")数据导入成功完成！"
else
echo "导入过程中发生错误，请检查日志！"
break;
fi
sleep 1 || break
#当数量不满足BATCH_SIZE的数量，但是while终止，需要进行中断.

if [[ $(cat "$BATCH_SIZE_FILE") -lt $BATCH_SIZE ]] ; then
echo "$(date) END"
exit 0
fi

((TOTAL_COUNTER++))


done

exec {FD_SQL}>&-
echo "[INFO] SQL_FILE $SQL_FILE 已经关闭($?)。"
mysql restore

TABLE_NAME AS '表名',
CREATE_TIME AS '创建时间',
TABLE_ROWS AS '行数',
DATA_LENGTH + INDEX_LENGTH AS '大小(字节)'
FROM information_schema.TABLES
WHERE TABLE_SCHEMA = 'database01' -- 替换为你的数据库名
AND CREATE_TIME > DATE_SUB(NOW(), INTERVAL 7 DAY)
ORDER BY CREATE_TIME DESC;

mysqldump备份数据库 导出数据库 + 包含 Function
mysqldump -u root -p \
--routines \
--events \
--triggers \
--databases your_database_name > db_backup.sql
默认情况下，mysqldump 不会导出存储过程和自定义函数。你必须显式添加 -R（或 --routines）参数
--routines (-R)：导出所有的 Stored Procedures 和 Functions。
--triggers：导出触发器（默认开启，建议显式声明）。
--events：导出定时事件。
所有非系统用户的创建与授权SQL的导出。 自动拼接出所有非系统用户的创建与授权 SQL , 导出后的 all_users_grants.sql 文件直接包含带有加密密码的 CREATE USER 和 GRANT 语句，迁移时在新库执行即可。
mysql -u root -p -B -N -e "
SELECT CONCAT('SHOW GRANTS FOR \'', user, '\'@\'', host, '\';')
FROM mysql.user
WHERE user NOT IN ('mysql.sys', 'mysql.session', 'mysql.infoschema', 'root');
" | mysql -u root -p -B -N | sed 's/$/;/' > all_users_grants.sql
