#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <locale.h>

#define MAX_LINE 8192

// 获取 UTF-8 字符字节长度
int utf8_byte_len(unsigned char ch) {
if ((ch & 0x80) == 0) return 1; // ASCII
if ((ch & 0xE0) == 0xC0) return 2; // 2字节 UTF-8
if ((ch & 0xF0) == 0xE0) return 3; // 3字节 UTF-8
if ((ch & 0xF8) == 0xF0) return 4; // 4字节 UTF-8
return 1;
}

// 跳过 ANSI 序列辅助函数（不修改外部指针）
int get_ansi_len(const char *str, int idx) {
int start = idx;
if (str[idx] == '\033' || str[idx] == '\x1B') {
idx++;
if (str[idx] == '[') {
idx++;
while (str[idx] != '\0' && !((str[idx] >= 'A' && str[idx] <= 'Z') ||
(str[idx] >= 'a' && str[idx] <= 'z') ||
str[idx] == '~')) {
idx++;
}
if (str[idx] != '\0') idx++;
return idx - start;
}
}
return 0;
}

void process_line(const char *src, char *dest) {
char buffer[MAX_LINE];
int cursor = 0; // 字节游标
int max_len = 0; // 记录到达过的最大长度

memset(buffer, 0, sizeof(buffer));

int i = 0;
while (src[i] != '\0') {
// 1. 过滤 ANSI 逃逸序列 (\033[...m, \033[?2004h 等)
int ansi_len = get_ansi_len(src, i);
if (ansi_len > 0) {
i += ansi_len;
continue;
}

// 2. 处理退格 \b
if (src[i] == '\b') {
if (cursor > 0) cursor--;
i++;
continue;
}

// 3. 处理回车 \r (关键修复：先探查 \r 之后跳过 ANSI 后的实际字符)
if (src[i] == '\r') {
int next_idx = i + 1;
// 探查跳过后续可能紧跟的 ANSI 序列
while (1) {
int a_len = get_ansi_len(src, next_idx);
if (a_len > 0) {
next_idx += a_len;
} else {
break;
}
}

// 如果 \r 后面（包含 ANSI 码之后）直接遇到了 \n 或字符串结尾，说明这只是普通的 DOS 换行，忽略 \r
if (src[next_idx] == '\n' || src[next_idx] == '\0') {
i++;
continue;
}

// 真正的终端重叠覆盖：光标复位回行首，并清空旧缓冲区，防止旧提示符残留
cursor = 0;
memset(buffer, ' ', max_len);
max_len = 0;
i++;
continue;
}

// 4. 处理换行 \n
if (src[i] == '\n') {
break;
}

// 5. 按字节/多字节字符逐个写入
int len = utf8_byte_len((unsigned char)src[i]);
for (int k = 0; k < len && src[i] != '\0'; k++) {
buffer[cursor++] = src[i++];
}

if (cursor > max_len) {
max_len = cursor;
}
}

// 清除行末尾无用的空格
int end_pos = max_len;
while (end_pos > 0 && buffer[end_pos - 1] == ' ') {
end_pos--;
}

buffer[end_pos] = '\0';
strcpy(dest, buffer);
}

int main(int argc, char *argv[]) {
setlocale(LC_ALL, "");

if (argc >= 3 && strcmp(argv[1], argv[2]) == 0) {
fprintf(stderr, "错误：输入文件和输出文件不能是同一个文件！\n");
return 1;
}

FILE *in = (argc > 1) ? fopen(argv[1], "r") : stdin;
if (!in) {
perror("无法打开输入文件");
return 1;
}

FILE *out = (argc > 2) ? fopen(argv[2], "w") : stdout;
if (!out) {
perror("无法打开输出文件");
if (in != stdin) fclose(in);
return 1;
}

char line_in[MAX_LINE];
char line_out[MAX_LINE];

while (fgets(line_in, sizeof(line_in), in)) {
process_line(line_in, line_out);
fprintf(out, "%s\n", line_out);
}

if (in != stdin) fclose(in);
if (out != stdout && out != stdin) fclose(out);

return 0;
}

expect将log写入文件 expect 脚本中，有多种方法可以将上次收到的数据写入文件 方法一：使用 log_file 命令（最简单） #!/usr/bin/expect # 开启日志记录，将所有交互内容写入文件 log_file output.log spawn ssh user@server expect "password:" send "mypassword\r" # 执行某个命令 expect "$ " send "ls -la\r" # 所有输出都会自动写入 output.log expect "$ " 方法二：手动捕获并写入文件 #!/usr/bin/expect spawn ssh user@server expect "password:" send "mypassword\r" # 等待命令执行并捕获输出 expect "$ " send "show version\r" # 捕获命令的输出内容（直到下一个提示符） expect -re "(.*)\$ " # 将捕获的内容写入文件 set f [open "command_output.txt" w] puts $f $expect_out(0,string) # 写入整个匹配的内容 # puts $f $expect_out(1,string) # 如果用了括号捕获，用这个 close $f send_user "Output saved to file\n" 方法三：精细化捕获特定内容 #!/usr/bin/expect spawn your-command # 捕获完整的交互过程 expect { "success" { # 捕获包含 "success" 之前的整个缓冲区 set output $expect_out(buffer) set f [open "success.log" w] puts $f $output close $f send_user "Success captured\n" } "failed" { set output $expect_out(buffer) set f [open "error.log" w] puts $f $output close $f send_user "Error captured\n" exit 1 } timeout { send_user "Timeout occurred\n" exit 1 } } 重要注意事项 $expect_out(buffer)：保存最近一次 expect 匹配的所有内容（包括匹配行之前的内容） $expect_out(0,string)：整个匹配的字符串（等同于 buffer） $expect_out(1,string)：第一个括号 () 捕获的内容 log_file 的选项： log_file -noappend file.log # 覆盖模式 log_file file.log # 追加模式 log_file -a file.log # 显式追加模式 expect_out(buffer)：这是你之前问过的“捕获输出”变量。它保存了 expect 命令匹配到的所有内容，通常用于保存日志或做进一步判断。 expect_out(0,string)：匹配到的整个模式。如果你用了正则表达式的括号捕获，(1,string) 就是第一个括号捕获的内容。 expect_out(spawn_id)：当前 spawn 进程的 ID，在同时管理多个进程时用于区分通信对象。
