# RDP-Shutdown-Notify.ps1
# 循环运行：每分钟检查一次 RDP 登录用户，向未通知过的用户发送关机提醒
# 终止时间：2026-10-28 17:00
# 已通知用户仅保存在内存变量 $notified 中，不使用文件

$Message = "系统即将于20261028 09:00 关机。"
$EndTime = Get-Date "2026-10-28 17:00:00"

# 内存中的已通知用户列表（脚本运行期间有效）
$notified = @()

Write-Host "脚本启动，终止时间：$EndTime" -ForegroundColor Cyan

while ((Get-Date) -lt $EndTime) {

# 获取当前 RDP 登录用户
$rdpUsers = @()
$quserOutput = quser 2>$null
if ($quserOutput) {
$rdpUsers = $quserOutput | Select-Object -Skip 1 | ForEach-Object {
$parts = $_ -split '\s+', 3
if ($parts.Count -ge 3 -and $parts[2] -match 'rdp') {
$parts[1]
}
} | Where-Object { $_ } | Select-Object -Unique
}

# 对未通知过的用户发送消息
foreach ($user in $rdpUsers) {
if ($notified -notcontains $user) {
msg $user $Message 2>$null
if ($LASTEXITCODE -eq 0) {
$notified += $user
Write-Host "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') 已通知用户：$user" -ForegroundColor Green
} else {
Write-Host "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') 通知失败：$user" -ForegroundColor Yellow
}
}
}

# 等待 1 分钟后再检查（临近终止时间时缩短睡眠）
$remain = ($EndTime - (Get-Date)).TotalSeconds
if ($remain -gt 0) {
Start-Sleep -Seconds ([Math]::Min(60, [int]$remain))
}
}

Write-Host "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') 已到达终止时间，脚本退出。" -ForegroundColor Cyan


按Session ID + 登录时间 + 用户名 通知，避免同一用户名多人登陆，退出导致的session id 重复问题

# RDP-Shutdown-Notify.ps1
# 循环运行：每分钟检查一次 RDP 会话，向未通知过的会话发送关机提醒
# 终止时间：2026-10-28 17:00
# 去重键：Session ID + 登录时间 + 用户名（三者完全一致才视为已通知）

$Message = "系统即将于20261028 09:00 关机。"
$EndTime = Get-Date "2026-10-28 17:00:00"

# 内存中已通知会话的键集合
$notifiedKeys = @()

Write-Host "脚本启动，终止时间：$EndTime" -ForegroundColor Cyan

while ((Get-Date) -lt $EndTime) {

# 解析 quser 输出，提取 RDP 会话信息
# quser 列（用户名可能带前导 > 表示当前会话）：
# 用户名 会话名 ID 状态 空闲时间 登录时间
# 注意：列之间可能有多空格，登录时间本身含空格，需按固定列位解析
$rdpSessions = @()
$quserOutput = quser 2>$null
if ($quserOutput) {
$rdpSessions = $quserOutput | Select-Object -Skip 1 | ForEach-Object {
$line = $_.TrimEnd()
# 用户名：第一段（去掉前导 >）
$user = ($line -split '\s+')[0].TrimStart('>')
# 会话名：需判断是否 rdp 会话
if ($line -match '\s+rdp') {
# 用正则提取各字段：用户名 会话名 ID 状态 空闲时间 登录时间
if ($line -match '^\s*>?\s*(\S+)\s+(\S+)\s+(\d+)\s+(\S+)\s+(\S+)\s+(.+)$') {
$sessionUser = $matches[1]
$sessionName = $matches[2]
$sessionId = [int]$matches[3]
$state = $matches[4]
$idleTime = $matches[5]
$logonTime = $matches[6].Trim()
[PSCustomObject]@{
User = $sessionUser
SessionId = $sessionId
LogonTime = $logonTime
Key = "$sessionId|$logonTime|$sessionUser"
}
}
}
}
}

# 对未通知过的会话发送消息
foreach ($session in $rdpSessions) {
if ($notifiedKeys -notcontains $session.Key) {
msg $session.SessionId $Message 2>$null
if ($LASTEXITCODE -eq 0) {
$notifiedKeys += $session.Key
Write-Host "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') 已通知：SessionID=$($session.SessionId) 用户=$($session.User) 登录时间=$($session.LogonTime)" -ForegroundColor Green
} else {
Write-Host "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') 通知失败：SessionID=$($session.SessionId) 用户=$($session.User)" -ForegroundColor Yellow
}
}
}

$remain = ($EndTime - (Get-Date)).TotalSeconds
if ($remain -gt 0) {
Start-Sleep -Seconds ([Math]::Min(60, [int]$remain))
}
}

Write-Host "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') 已到达终止时间，脚本退出。" -ForegroundColor Cyan
